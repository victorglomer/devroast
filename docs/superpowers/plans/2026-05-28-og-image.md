# OG Image Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Generate Open Graph images for roast result pages using Takumi, following the Pencil "Screen 4 - OG Image" design.

**Architecture:** Route handler at `/api/og?submissionId=xxx` fetches roast data from DB, loads JetBrains Mono from Google Fonts, renders a React component (inline styles matching Pencil design) via Takumi's `ImageResponse`, and returns a 1200×630 PNG with `Cache-Control: immutable`.

**Tech Stack:** Next.js 16 App Router, `takumi-js` + `@takumi-rs/core`, Google Fonts CSS API, Drizzle ORM

### Task 3: Create Font Loading Utility

**Files:**
- Create: `src/lib/og-fonts.ts`

- [ ] **Step 1: Write font loader**

Create `src/lib/og-fonts.ts`:

```typescript
import type { Font } from "takumi-js";

interface FontFaceDecl {
  weight: number;
  style: string;
  url: string;
}

function parseGoogleFontsCss(css: string): FontFaceDecl[] {
  const decls: FontFaceDecl[] = [];
  const blockRegex = /@font-face\s*\{([^}]+)\}/g;
  let blockMatch: RegExpExecArray | null;

  while ((blockMatch = blockRegex.exec(css)) !== null) {
    const block = blockMatch[1];
    const props: Record<string, string> = {};
    const propRegex = /([\w-]+)\s*:\s*(.+?)\s*;/g;
    let propMatch: RegExpExecArray | null;
    while ((propMatch = propRegex.exec(block)) !== null) {
      props[propMatch[1].trim()] = propMatch[2].trim();
    }
    const srcMatch = props.src?.match(/url\(([^)]+)\)/);
    if (srcMatch && props["font-family"]?.includes("JetBrains Mono")) {
      decls.push({
        weight: Number(props["font-weight"]) || 400,
        style: props["font-style"] || "normal",
        url: srcMatch[1],
      });
    }
  }
  return decls;
}

export async function loadJetBrainsMonoFonts(): Promise<Font[]> {
  const cssUrl =
    "https://fonts.googleapis.com/css2?family=JetBrains+Mono:wght@400;500;700;900&display=swap";
  const css = await fetch(cssUrl).then((r) => r.text());
  const decls = parseGoogleFontsCss(css);

  const fonts = await Promise.all(
    decls.map(async (decl) => {
      const data = await fetch(decl.url).then((r) => r.arrayBuffer());
      return {
        name: "JetBrains Mono",
        data,
        weight: decl.weight,
        style: decl.style,
      } satisfies Font;
    }),
  );

  return fonts;
}
```

### Task 4: Create OG Image API Route

**Files:**
- Create: `src/app/api/og/route.tsx`

- [ ] **Step 1: Write the route handler**

Create `src/app/api/og/route.tsx`:

```typescript
import { ImageResponse } from "takumi-js/response";
import { OgImage } from "@/components/og-image";
import { getSubmissionWithRoast } from "@/db/queries";
import { loadJetBrainsMonoFonts } from "@/lib/og-fonts";

export const runtime = "nodejs";

export async function GET(request: Request) {
  const { searchParams } = new URL(request.url);
  const submissionId = searchParams.get("submissionId");

  if (!submissionId) {
    return fallbackResponse();
  }

  try {
    const result = await getSubmissionWithRoast(submissionId);

    if (!result) {
      return fallbackResponse();
    }

    const score = Number(result.score);
    const [fonts] = await Promise.all([
      loadJetBrainsMonoFonts(),
    ]);

    const response = new ImageResponse(
      <OgImage
        score={score}
        verdict={result.verdict}
        roastTitle={result.roastTitle}
        language={result.language}
        lineCount={result.lineCount}
      />,
      {
        width: 1200,
        height: 630,
        fonts,
      },
    );

    response.headers.set(
      "Cache-Control",
      "public, max-age=31536000, immutable",
    );

    return response;
  } catch {
    return fallbackResponse();
  }
}

function fallbackResponse(): Response {
  const response = new ImageResponse(
    <div
      style={{
        width: "100%",
        height: "100%",
        display: "flex",
        flexDirection: "column",
        justifyContent: "center",
        alignItems: "center",
        gap: "16px",
        backgroundColor: "#0A0A0A",
      }}
    >
      <div style={{ display: "flex", gap: "8px" }}>
        <span
          style={{
            fontFamily: "JetBrains Mono",
            fontSize: "32px",
            fontWeight: 700,
            color: "#10B981",
          }}
        >
          &gt;
        </span>
        <span
          style={{
            fontFamily: "JetBrains Mono",
            fontSize: "28px",
            fontWeight: 500,
            color: "#FAFAFA",
          }}
        >
          devroast
        </span>
      </div>
      <span
        style={{
          fontFamily: "Geist",
          fontSize: "18px",
          color: "#6B7280",
        }}
      >
        Code Review Roast
      </span>
    </div>,
    { width: 1200, height: 630 },
  );

  response.headers.set(
    "Cache-Control",
    "public, max-age=31536000, immutable",
  );

  return response;
}
```
